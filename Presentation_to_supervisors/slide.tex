\section{Introduction}
% =====================================================================

\begin{frame}{What is this about?}
    \begin{itemize}
        \item \textbf{SG52}: Subgroup 52 of the OECD-NEA Working Party on International Nuclear Data
              Evaluation Co-operation (WPEC).
        \item Topic: \textbf{nuclear data adjustment}, i.e.\ updating evaluated nuclear data with measurements.
        \item Nuclide and range: \textbf{Pu-239}, fast neutron energies.
        \item Method: generalized linear least squares (\textbf{GLLS}).
        \item The coordinators send data \emph{releases}; every participant returns results in a common
              template; the coordinators compare the participants.
        \item Participants so far: EPFL, SCK CEN, UPM and UU (us).
    \end{itemize}
\end{frame}

\begin{frame}{The goal}
    \begin{block}{From the SG52 specification \cite{sg52spec}}
        \small
        Reconstruct a modern evaluation, ENDF/B-VIII.1 \cite{nobre2026endf}, from its predecessor,
        ENDF/B-VIII.0 \cite{brown2018endf}, using the same experimental data as the evaluators.
    \end{block}
    \vspace{4mm}
    \centering
    \begin{tikzpicture}[node distance=5mm and 6mm, font=\scriptsize]
        \node[box=20mm] (old) {\textbf{ENDF/B-VIII.0}\\Pu-239 (prior)};
        \node[box=20mm, below=of old] (exp) {\textbf{Experiments}\\integral and differential};
        \node[result=15mm, anchor=west] (glls) at ($(old.east)!0.5!(exp.east) + (6mm,0)$) {\textbf{GLLS}\\adjustment};
        \node[result=17mm, right=of glls] (adj) {\textbf{Adjusted}\\Pu-239};
        \node[box=20mm, right=15mm of adj] (new) {\textbf{ENDF/B-VIII.1}\\Pu-239};
        \draw[arr] (old.east) -- (glls);
        \draw[arr] (exp.east) -- (glls);
        \draw[arr] (glls) -- (adj);
        \draw[arr, dashed, <->] (adj) -- (new) node[lbl, midway, below=1mm] {should agree};
    \end{tikzpicture}\\[5mm]
    \small
    \ldots\ without weakening the criticality, reactor and shielding results that already work.
\end{frame}

\begin{frame}{Workflow: one reproducible pipeline per exercise}
    \centering
    \begin{tikzpicture}[node distance=6mm and 7mm, font=\scriptsize]
        \node[box=20mm] (rel) {Release data\\covariances, measurements, sensitivities};
        \node[box=19mm, right=of rel, yshift=9mm] (pri) {Prior builder (Python)\\$\mu_i$, $\Sigma_i$};
        \node[box=19mm, right=of rel, yshift=-9mm] (obs) {Observation builder (Python)\\$\Delta$, $S$, $\Sigma_E$};
        \node[result=19mm, right=of pri, yshift=-9mm] (nda) {\texttt{nda}: GLLS in C++20\\$\mu_f$, $\Sigma_f$};
        \node[box=19mm, right=of nda, yshift=9mm] (dia) {Diagnostics\\checks and plots};
        \node[box=19mm, right=of nda, yshift=-9mm] (sub) {NEA template\\(Excel)};
        \draw[arr] (rel.east) -- (pri.west);
        \draw[arr] (rel.east) -- (obs.west);
        \draw[arr] (pri) -- (obs);
        \draw[arr] (pri.east) -- (nda.west);
        \draw[arr] (obs.east) -- (nda.west);
        \draw[arr] (nda.east) -- (dia.west);
        \draw[arr] (nda.east) -- (sub.west);
    \end{tikzpicture}
    \vspace{4mm}
    \begin{itemize}
        \small
        \item The C++ GLLS program is generic; everything specific to SG52 lives in the Python steps.
        \item Every reported number is computed from the release files, and reruns give identical results.
    \end{itemize}
\end{frame}

    \begin{tikzpicture}[x=1mm, y=1mm, font=\scriptsize,

        % shapes show matrix dimensions: n states (18 mm) against m observations (6 mm)
        \filldraw[fill=uppsalared!25, draw=uppsalared] (8.75,2) rectangle (11.25,20);
        \filldraw[fill=uppsalared!25, draw=uppsalared] (22,2) rectangle (40,20);

        \filldraw[fill=uppsalared!55, draw=uppsalared] (91,8) rectangle (97,14);
        % descriptions
        \node[desc] at (10,0) {$\mu_i$\\\textbf{prior mean}\\ENDF/B-VIII.0 values\\size $n$};
        \node[desc] at (31,0) {$\Sigma_i$\\\textbf{prior covariance}\\how uncertain and correlated the prior is\\$n \times n$};
        \node[desc] at (52,0) {$S$\\\textbf{sensitivity}\\how each calculated response moves with each state\\$m \times n$};
        \node[desc] at (73,0) {$\Delta = E - C$\\\textbf{discrepancy}\\measured minus calculated\\size $m$};
        \node[desc] at (94,0) {$\Sigma_E$\\\textbf{observation covariance}\\measurement and calculation uncertainty\\$m \times m$};

    $n = \ExVal{1}{n}$ states; $m$ = \ExVal{1}{m} (Ex1) to \ExVal{4}{m} (Ex4) observations.\\[2pt]
    The release gives relative values; we convert them to absolute units first.
\end{frame}

\begin{frame}{Putting it together: the GLLS update}
    \centering
    \begin{tikzpicture}[node distance=2mm and 5mm, font=\scriptsize]
        \node[box=10mm] (s) {$S$};
        \node[box=10mm, above=of s] (si) {$\Sigma_i$};
        \node[box=10mm, below=of s] (se) {$\Sigma_E$};
        \node[box=23mm, right=of s] (v) {$V = S \Sigma_i S^\top + \Sigma_E$\\[2pt]
            total uncertainty of the discrepancy};
        \node[box=23mm, right=of v] (k) {$K = \Sigma_i S^\top V^{-1}$\\[2pt]
            gain: how far each state moves per unit discrepancy};
        \node[result=26mm, right=of k, yshift=9mm] (muf) {$\mu_f = \mu_i + K \Delta$\\[2pt]
            \textbf{posterior mean}};
        \node[result=26mm, right=of k, yshift=-9mm] (sf) {$\Sigma_f = \Sigma_i - K S \Sigma_i$\\[2pt]
            \textbf{posterior covariance}};
        \draw[arr] (si.east) -- (v.west);
        \draw[arr] (s) -- (v);
        \draw[arr] (se.east) -- (v.west);
        \draw[arr] (v) -- (k);
        \draw[arr] (k.east) -- (muf.west);
        \draw[arr] (k.east) -- (sf.west);
    \end{tikzpicture}\\[6mm]
    \small
    The data pull the mean toward agreement with the measurements and shrink the covariance.\\[4pt]
    Reported relative to the prior mean: adjustment $(\mu_f - \mu_i)/\mu_i$,
    covariance $\Sigma_f / (\mu_i \mu_i^\top)$, element by element.
\end{frame}

\section{Discussion}
% =====================================================================

\begin{frame}{How we compare with the other participants}
    \begin{itemize}
        \item \textbf{Ex1--3}: all four participants' adjustments and uncertainties lie on top of each other
              in every coordinator plot we inspected.
        \item \textbf{Ex4}: EPFL, SCK CEN and UU still overlap; UPM differs, most visibly in MT452 $\bar\nu$
              (slightly in MT4 and MT18). The coordinators have not stated a cause.
        \item The release 4 specification now spells out our Ex4 treatment: compare C and E on MT456,
              apply the adjustment to MT452 as if the data were on it.
    \end{itemize}
    \vspace{2mm}
    \begin{block}{Takeaway}
        Independent codes and workflows reproduce one another for the integral exercises: an external
        cross-check of our absolute-form GLLS and the relative/absolute conversions.
    \end{block}
\end{frame}

\begin{frame}{Points worth flagging}
    \begin{itemize}
        \small
        \item \textbf{Marini $\bar\nu$ lands below the posterior.} The \ExVal{4}{mariniN} points are strongly correlated
              (mean off-diagonal $\rho$ = \ExVal{4}{mariniRhoMean}; one eigenmode carries \ExVal{4}{mariniTopMode}\,\%
              of the correlation matrix), and the prior disagrees with their shape.
              GLLS fits the shape and moves the common level, so all posterior C/E end below 1.
              We report this as it is and do not correct it.
        \item \textbf{Prior correlation plots differ in below-threshold groups.} \ExVal{1}{zeroMean} states have
              zero prior mean (below-threshold groups and a few low-energy PFNS bins); the released relative
              covariance carries tiny non-zero entries there. The absolute
              conversion sets them to zero, so our plots are blank where the reference shows noise
              (no effect on the adjustment).
        \item \textbf{GLLS limits.} No PFNS normalization or positivity constraint: we flag deviations, never clip.
              Adjusted responses are linearized predictions.
    \end{itemize}
\end{frame}
